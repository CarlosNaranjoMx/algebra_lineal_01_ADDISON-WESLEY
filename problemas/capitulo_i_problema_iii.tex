\section*{Capítulo I, Problema III}

\textbf{Problema:}

3. Usando solo las cuatro propiedades del producto escalar, verifica en detalle las identidades dadas en el texto para $(A + B)^2$ y $(A - B)^2$.

\textbf{Solución:}

Para verificar las identidades $(A + B)^2$ y $(A - B)^2$ usando las propiedades del producto escalar, procedemos de la siguiente manera:

\begin{itemize}
    \item \textbf{Identidad 1: $(A + B)^2$}
    \begin{align*}
        (A + B)^2 &= (A + B) \cdot (A + B) \\
        &= A \cdot A + 2(A \cdot B) + B \cdot B \\
        &= \|A\|^2 + 2(A \cdot B) + \|B\|^2.
    \end{align*}

    \item \textbf{Identidad 2: $(A - B)^2$}
    \begin{align*}
        (A - B)^2 &= (A - B) \cdot (A - B) \\
        &= A \cdot A - 2(A \cdot B) + B \cdot B \\
        &= \|A\|^2 - 2(A \cdot B) + \|B\|^2.
    \end{align*}
\end{itemize}

Ambas identidades se verifican utilizando las propiedades del producto escalar: conmutatividad, distributividad, y la relación $A \cdot A = \|A\|^2$.

\textbf{Problema:}

4. ¿Cuál de los siguientes pares de vectores son perpendiculares?

(a) $(1, -1, 1)$ y $(2, 1, 5)$\\
(b) $(1, -1, 1)$ y $(2, 3, 1)$\\
(c) $(-5, 2, 7)$ y $(3, -1, 2)$\\
(d) $(\pi, 2, 1)$ y $(2, -\pi, 0)$\\

\textbf{Solución:}

Dos vectores son perpendiculares si su producto escalar es igual a cero. Calculamos el producto escalar para cada par:

\begin{itemize}
    \item \textbf{(a)} $(1, -1, 1) \cdot (2, 1, 5) = 1(2) + (-1)(1) + 1(5) = 2 - 1 + 5 = 6 \neq 0$. No son perpendiculares.
    \item \textbf{(b)} $(1, -1, 1) \cdot (2, 3, 1) = 1(2) + (-1)(3) + 1(1) = 2 - 3 + 1 = 0$. Son perpendiculares.
    \item \textbf{(c)} $(-5, 2, 7) \cdot (3, -1, 2) = (-5)(3) + 2(-1) + 7(2) = -15 - 2 + 14 = -3 \neq 0$. No son perpendiculares.
    \item \textbf{(d)} $(\pi, 2, 1) \cdot (2, -\pi, 0) = \pi(2) + 2(-\pi) + 1(0) = 2\pi - 2\pi + 0 = 0$. Son perpendiculares.
\end{itemize}

Por lo tanto, los pares de vectores perpendiculares son \textbf{(b)} y \textbf{(d)}.

\textbf{Problema:}

5. Considera funciones continuas en el intervalo $[-1, 1]$. Define el producto escalar de dos funciones $f, g$ como

\[
    \langle f, g \rangle = \int_{-1}^{1} f(x)g(x) \, dx.
\]

Verifica que las cuatro reglas para un producto escalar se cumplen, es decir, demuestra que:

\begin{itemize}
    \item \textbf{SP 1.} $\langle f, g \rangle = \langle g, f \rangle$.
    \item \textbf{SP 2.} $\langle f, g + h \rangle = \langle f, g \rangle + \langle f, h \rangle$.
    \item \textbf{SP 3.} $\langle cf, g \rangle = c \langle f, g \rangle$.
    \item \textbf{SP 4.} Si $f = 0$, entonces $\langle f, f \rangle = 0$ y si $f \neq 0$, entonces $\langle f, f \rangle > 0$.
\end{itemize}

\textbf{Solución:}

\begin{itemize}
    \item \textbf{SP 1.} Por la propiedad conmutativa de la multiplicación:
    \[
        \langle f, g \rangle = \int_{-1}^{1} f(x)g(x) \, dx = \int_{-1}^{1} g(x)f(x) \, dx = \langle g, f \rangle.
    \]

    \item \textbf{SP 2.} Por la propiedad distributiva de la suma:
    \[
        \langle f, g + h \rangle = \int_{-1}^{1} f(x)(g(x) + h(x)) \, dx = \int_{-1}^{1} f(x)g(x) \, dx + \int_{-1}^{1} f(x)h(x) \, dx = \langle f, g \rangle + \langle f, h \rangle.
    \]

    \item \textbf{SP 3.} Por la propiedad asociativa de la multiplicación:
    \[
        \langle cf, g \rangle = \int_{-1}^{1} (cf(x))g(x) \, dx = c \int_{-1}^{1} f(x)g(x) \, dx = c \langle f, g \rangle.
    \]

    \item \textbf{SP 4.} Si $f = 0$, entonces $f(x) = 0$ para todo $x$ y
    \[
        \langle f, f \rangle = \int_{-1}^{1} f(x)f(x) \, dx = \int_{-1}^{1} 0 \, dx = 0.
    \]
    Si $f \neq 0$, entonces $f(x) \neq 0$ en al menos un punto del intervalo $[-1, 1]$, y como $f(x)^2 \geq 0$ para todo $x$, se tiene que
    \[
        \langle f, f \rangle = \int_{-1}^{1} f(x)^2 \, dx > 0.
    \]
\end{itemize}
